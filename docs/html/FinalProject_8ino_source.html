101// Name: moisturePercentFromRaw
102// Description: Maps raw ADC reading to 0..100% given DRY/WET calibration.
103static int moisturePercentFromRaw(int raw, int dryRaw, int wetRaw) {
104 // If calibration values are equal return 0% just to be safe
105 if (dryRaw == wetRaw) return 0;
106
107 float pct;
108 // If the reading is more dry or wet than the other
109 // Calculate the percent for moisture
110 if (dryRaw > wetRaw) {
111 pct = ((float)(dryRaw - raw) / (float)(dryRaw - wetRaw)) * 100.0f;
112 } else {
113 pct = ((float)(raw - dryRaw) / (float)(wetRaw - dryRaw)) * 100.0f;
114 }
115
116 // Ensures noise or out-of-range values do not go below 0% or above 100%
117 if (pct < 0.0f) pct = 0.0f;
118 if (pct > 100.0f) pct = 100.0f;
119
120 // Convers float to int, rouning to nearest whole percent
121 return (int)(pct + 0.5f);
122}

124// Name: pingUltrasonicCM
125// Description: Sends one HC-SR04-style ping and returns distance in cm.
126static float pingUltrasonicCM(void) {
127 // Sequence:
128 // LOW to reset
129 // Sending 10 microseconds HIGH pulse to the TRIG pin to trigger the sensor
130 // Back to low to end pulse
131 digitalWrite(ULTRA_TRIG_PIN, LOW); delayMicroseconds(2);
132 digitalWrite(ULTRA_TRIG_PIN, HIGH); delayMicroseconds(10);
133 digitalWrite(ULTRA_TRIG_PIN, LOW);
134
135 // pulseIn() waits for the ECHO pin to go HIGH, then times how long it stays HIGH
136 // The duration is the round-trip travel time of the ultrasonic sound wave
137 unsigned long dur = pulseIn(ULTRA_ECHO_PIN, HIGH, ECHO_TIMEOUT_US);
138
139 // If no echo is received before timeout, return a sentinel value (invalid)
140 if (dur == 0) return -1.0f;
141
142 // Convert time to distance
143 return dur * US_TO_CM;
144}

146// Name: ctrlTickCb
147// Description: Callback is trigged by the timer at 50 Hz
148// and signals TaskPump via a queue,
149// Think of it like a timer alarm bell where it rings
150// and drops a note into the queue, the task listens for
151// the notes and knows when it is time to do a control cycle.
152// For the pump, this is to ensure that the system is fast enough
153// to check soil moisture and ultrasonic regularly, reacting quickly
154// when soil is dry. Every 20ms, the system decides if the pump should be ON/OFF.
155static void IRAM_ATTR ctrlTickCb(void *arg) {
156 uint8_t b = 1;
157 // Ensure the queue was created successfully
158 if (qCtrlTick != NULL) {
159 // Pushes the byte into the control tick queue
160 // xQueueSendFromISR to put data into queue but in a safe way from an interrupt
161 // This wakes up TaskPump, which is waiting on qCtrlTick
162 xQueueSendFromISR(qCtrlTick, &b, NULL);
163 }
164}
165
166// Name: uiTickCb
167// Description: 64 Hz timer callback -> posts a byte into qUITick.
168// Same style as the TaskPump but for LCDTask and faster.
169// Keeps the LCD feeling smooth, without visible lag.
170static void IRAM_ATTR uiTickCb(void *arg) {
171 uint8_t b = 1;
172 if (qUITick != NULL) {
173 xQueueSendFromISR(qUITick, &b, NULL);
174 }
175}

179// Name: TaskPump
180// Description: Reads sensor/button states, controls relay + LED, AUTO watering,
181// and publishes a snapshot to the LCD task via qStatus.
182// Notes:
183// - Priority over pump: SUSPEND (OFF) -> MANUAL (ON) -> AUTO (ON/OFF).
184// - Again, when AUTO starts, pump remains ON for at least 10 seconds, then waits for 90 seconds before checking again.
185// - Idle timer starts once water pump finishes executing to show when it was lasted watered.
186void TaskPump(void *pv) {
187 (void)pv;
188
189 // Attenuation = how much the input voltage is "scaled down" before conversion for soil moisture sensor pin
190 // ADC_11db - input voltage range 0.0 - 3.6V
191 analogSetPinAttenuation(SENSOR_PIN, ADC_11db);
192
193 // Button debounce state
194 int lastManual = HIGH; // last read of manual btn
195 int manualStable = HIGH; // debounced manual
196 uint32_t manualAt = 0; // last change time
197
198 int lastSuspend = HIGH; // last read of suspend btn
199 int suspendStable = HIGH; // debounced suspend
200 uint32_t suspendAt = 0; // last change time
201
202 // Suspend flags
203 uint8_t suspendedButton = 0; // user toggle
204 uint8_t suspendedAuto = 0; // tank empty
205
206 // Pump + idle timer
207 uint8_t pumpOn = 0;
208 uint8_t lastPumpOn = 0;
209 uint8_t first = 1;
210 TickType_t offStart = 0;
211
212 // AUTO
213 uint8_t inAutoBurst = 0;
214 uint32_t autoOnUntilMs = 0;
215 uint32_t soakUntilMs = 0;
216
217 // Reservoir distance tracker
218 float lastDistCM = -1.0f;
219
220 // Snapshot to LCD
221 status_t s; // struct
222 memset(&s, 0, sizeof(s)); // fills all fields of the struct with 0
223 s.distanceX100 = -1; // marks distance as invalid
224
225 uint8_t tick;
226 TickType_t fastScanDelay = pdMS_TO_TICKS(5); // ~200 Hz button scan
227
228 while (1) {
229 // Current time in milliseconds
230 uint32_t nowms = millis();
231
232 // Debounce: Manual (hold)
233 int mRead = digitalRead(BTN_MANUAL_PIN); // sample button
234 if (mRead != lastManual) { // state changes?
235 manualAt = nowms; // start/restart the debounce timer
236 }
237 if ((nowms - manualAt) >= DEBOUNCE_MS) { // state stayed unchanged long enough?
238 if (manualStable != mRead) { // is it a new stable value?
239 manualStable = mRead; // accepts as the debounced state
240 }
241 }
242 lastManual = mRead; // remember for next iteration
243
244 // Debounce: Suspend (toggle on press)
245 int sRead = digitalRead(BTN_SUSP_PIN);
246 if (sRead != lastSuspend) {
247 suspendAt = nowms;
248 }
249 if ((nowms - suspendAt) >= DEBOUNCE_MS) {
250 if (suspendStable != sRead) {
251 // falling edge (pressed = LOW) toggles suspend
252 if (suspendStable == HIGH) { // previously HIGH (not pressed)
253 if (sRead == LOW) { // now LOW (pressed)
254 // If it was 0 (auto active), it becomes 1 (auto suspended)
255 if (suspendedButton == 0) suspendedButton = 1;
256 // If it was 1 (auto suspended), it becomes 0 (auto active)
257 else suspendedButton = 0;
258 }
259 }
260 suspendStable = sRead;
261 }
262 }
263 lastSuspend = sRead;
264
265 // 50 Hz control tick
266 // Only execute the control loop when the timer ISR drops a byto into qCtrlTick
267 if (xQueueReceive(qCtrlTick, &tick, fastScanDelay) == pdTRUE) {
268 // Read soil moisture and turn into a %
269 int raw = analogRead(SENSOR_PIN);
270 if (raw > DRY_RAW) raw = DRY_RAW;
271 if (raw < WET_RAW) raw = WET_RAW;
272 int moisturePct = moisturePercentFromRaw(raw, DRY_RAW, WET_RAW);
273
274 // Read ultrasonic tank distance and keep last value
275 float dcm = pingUltrasonicCM();
276 if (dcm > 0.0f) {
277 lastDistCM = dcm;
278 }
279
280 // Detect if tank is empty and set auto-suspend
281 uint8_t tankEmpty = 0;
282 if (lastDistCM > 0.0f) {
283 if (lastDistCM >= EMPTY_DISTANCE_CM) {
284 tankEmpty = 1;
285 }
286 }
287 suspendedAuto = tankEmpty;
288
289 // Drive the status LED whether the button is pressed or tank is empty
290 if (tankEmpty == 1 || suspendedButton == 1) {
291 digitalWrite(WATER_LOW_PIN, HIGH);
292 } else {
293 digitalWrite(WATER_LOW_PIN, LOW);
294 }
295
296 // Combined suspend sources
297 // Single flag that says whether any suspension is active
298 uint8_t suspended = 0;
299 if (suspendedButton == 1) suspended = 1; // user toggle
300 if (suspendedAuto == 1) suspended = 1; // tank empty
301
302 // -Decide pump state with priority: Suspend -> Manual -> Auto
303 uint8_t prev = pumpOn;
304
305 if (suspended == 1) {
306 // Force OFF and cancel any AUTO
307 pumpOn = 0;
308 inAutoBurst = 0;
309 digitalWrite(RELAY_PIN, HIGH);
310 } else if (manualStable == LOW) {
311 // Manual hold: force ON (cancel AUTO timing)
312 pumpOn = 1;
313 inAutoBurst = 0;
314 digitalWrite(RELAY_PIN, LOW);
315 } else {
316 // AUTO section
317 if (inAutoBurst == 1) {
318 // Enforce minimum ON duration regardless of threshold
319 if (nowms < autoOnUntilMs) {
320 pumpOn = 1;
321 digitalWrite(RELAY_PIN, LOW);
322 } else {
323 // Finish -> OFF until 90 seconds elasped from current time
324 inAutoBurst = 0;
325 pumpOn = 0;
326 digitalWrite(RELAY_PIN, HIGH);
327 soakUntilMs = nowms + SOAK_MS;
328 }
329 }
330
331 // If the state is not in AUTO
332 if (inAutoBurst == 0) {
333 // Pump stays OFF
334 uint8_t autoAllowed = 1;
335 if (nowms < soakUntilMs) autoAllowed = 0;
336
337 // READY state
338 if (autoAllowed == 1) {
339 // Pump ON
340 if (moisturePct < MOISTURE_ON_THRESHOLD) {
341 inAutoBurst = 1;
342 autoOnUntilMs = nowms + MIN_ON_MS;
343 pumpOn = 1;
344 digitalWrite(RELAY_PIN, LOW);
345 } else {
346 // READY but not dry enough
347 pumpOn = 0;
348 digitalWrite(RELAY_PIN, HIGH);
349 }
350 } else {
351 // Force pump OFF regardless of moisture
352 pumpOn = 0;
353 digitalWrite(RELAY_PIN, HIGH);
354 }
355 }
356 }
357
358 // Idle timer logic (seconds since last watering OFF)
359 // Record the current pump state
360 // If pump starts OFF, record the tick count as the beginning of idle period
361 if (first == 1) {
362 first = 0;
363 lastPumpOn = pumpOn;
364 if (pumpOn == 0) {
365 offStart = xTaskGetTickCount();
366 }
367 }
368
369 // Detect OFF to ON transition (pump just started)
370 // Reset idle counter when it starts watering
371 if (prev == 0) {
372 if (pumpOn == 1) {
373 // OFF -> ON
374 s.idleSeconds = 0;
375 }
376 }
377
378 // Detects ON to OFF transition (watering finished)
379 // Reset idle counter again and mark this time as the new offStart
380 if (prev == 1) {
381 if (pumpOn == 0) {
382 // ON -> OFF
383 offStart = xTaskGetTickCount();
384 s.idleSeconds = 0;
385 }
386 }
387
388 // While pump is OFF: compute elasped time since it lasted turned off
389 // While it is ON: report 0
390 // Update lastPumpOn for the next cycle's transition check
391 if (pumpOn == 0) {
392 TickType_t nowt = xTaskGetTickCount();
393 // Convert ticks to ms to seconds
394 uint32_t ms = (uint32_t)((nowt - offStart) * portTICK_PERIOD_MS);
395 s.idleSeconds = ms / 1000U;
396 } else {
397 s.idleSeconds = 0;
398 }
399 lastPumpOn = pumpOn;
400
401 // Record soil moisture, pump state, and suspend flag
402 s.moisturePct = moisturePct;
403 s.pumpOn = pumpOn;
404 s.suspended = suspended;
405
406 // Store tank distance
407 // If no valid reading, set 01
408 // Flag if tank is empty
409 if (lastDistCM > 0.0f) {
410 int x100 = (int)(lastDistCM * 100.0f + 0.5f);
411 s.distanceX100 = x100;
412 } else {
413 s.distanceX100 = -1;
414 }
415 s.tankEmpty = tankEmpty;
416
417 // Push this fresh snapshot to qStatus
418 if (qStatus != NULL) {
419 // Ensure the LCD task always has the latest status
420 xQueueOverwrite(qStatus, &s);
421 }
422 }
423 }
424}

426// Name: TaskLCD
427// Description: display task pinned to the other core
428// - initializes the LCD and custom icons
429// - waits for a 64 Hz UI tick
430// - pulls the latest snapshot from the queue
431// - only redraws parts of the screen that actually changed
432// Note: Top row: moisture (%) and timer (--:--:--); Bottom row: tank (--:--cm) and pump (ON/OFF)
433void TaskLCD(void *pv) {
434 (void)pv;
435
436 // Initialize LCD + icons
437 lcd.clear();
438 lcd.createChar(0, ICON_DROP);
439 lcd.createChar(1, ICON_CLOCK);
440 lcd.createChar(2, ICON_TANK);
441 lcd.createChar(3, ICON_PUMP);
442 vTaskDelay(pdMS_TO_TICKS(50));
443
444 // Cached state for redraw minimization avoiding constant rewrite the same characters
445 status_t s; // struct
446 memset(&s, 0, sizeof(s)); // initialize all fields to 0
447 s.moisturePct = -1;
448 s.distanceX100 = -1;
449
450 // These hold the last values actually drawn to the LCD
451 // Initialize to "impossible" values to guarntee the very first update forces a redraw
452 int lastPct = -1;
453 uint32_t lastSecs = 0xFFFFFFFF;
454 int lastDist = -999999;
455 uint8_t lastPumpOn = 0xFF;
456
457 // Clear bottom once
458 lcd.setCursor(0, 1);
459 lcd.print(" ");
460
461 uint8_t tick;
462 while (1) {
463 // wait for UI tick
464 if (xQueueReceive(qUITick, &tick, portMAX_DELAY) == pdTRUE) {
465 // pull latest snapshot (non-blocking)
466 // reading the latest system state from qStatus
467 status_t tmp;
468 if (qStatus != NULL) {
469 if (xQueueReceive(qStatus, &tmp, 0) == pdTRUE) {
470 s = tmp;
471 }
472 }
473
474 // TOP ROW
475 // Only redraw if moisture or timer changed
476 if (s.moisturePct != lastPct || s.idleSeconds != lastSecs) {
477 lcd.setCursor(0, 0);
478 lcd.print(" ");
479
480 // Ensure % stays in 0-100 range
481 int pct = s.moisturePct;
482 if (pct < 0) pct = 0;
483 if (pct > 100) pct = 100;
484
485 // Compute hrs:mins:secs
486 uint32_t secs = s.idleSeconds;
487 uint32_t hh = secs / 3600U;
488 uint32_t mm = (secs / 60U) % 60U;
489 uint32_t ss = secs % 60U;
490
491 // Compose the row with icons and spacing
492 // column 0: draw custom char ICON_DROP at index 0
493 // column 1: space
494 // Cursor col is incremented each time so the next write is placed correctly
495 int col = 0;
496 lcd.setCursor(col, 0); lcd.write((uint8_t)0); col++; // drop
497 lcd.setCursor(col, 0); lcd.print(' '); col++;
498
499 // Print moisture percent
500 // Format the number into string
501 char pctbuf[4];
502 if (pct == 100) {
503 strcpy(pctbuf, "100");
504 } else {
505 snprintf(pctbuf, sizeof(pctbuf), "%d", pct);
506 }
507 lcd.setCursor(col, 0); lcd.print(pctbuf); col += (int)strlen(pctbuf);
508
509 // Add percent sign, space, clock icon
510 lcd.setCursor(col, 0); lcd.print('%'); col++;
511 lcd.setCursor(col, 0); lcd.print(' '); col++;
512
513 lcd.setCursor(col, 0); lcd.write((uint8_t)1); col++; // clock
514 lcd.setCursor(col, 0); lcd.print(' '); col++;
515
516 // Print timer
517 char timebuf[9];
518 snprintf(timebuf, sizeof(timebuf), "%02u:%02u:%02u",
519 (unsigned)hh, (unsigned)mm, (unsigned)ss);
520
521 // special layout case for 100% to prevent overflow
522 if (pct == 100) {
523 col = 8; // keep within 16 cols when 3-digit percent
524 }
525 lcd.setCursor(col, 0); lcd.print(timebuf);
526
527 // Update "last drawn" cache
528 lastPct = s.moisturePct;
529 lastSecs = s.idleSeconds;
530 }
531
532 // Bottom Row
533 // Only redraw when distance of pump state changes
534 if (s.distanceX100 != lastDist || s.pumpOn != lastPumpOn) {
535 // Format the distance string
536 char distStr[7]; // 6 chars + NUL
537
538 // Valid distance
539 // Split into integer part and two digit fractional part
540 if (s.distanceX100 >= 0) {
541 int whole = s.distanceX100 / 100; // integer cm
542 int frac = s.distanceX100 % 100; // hundredths
543
544 // Show a clmap marker if it exceeds 6 chars
545 // Otherwise, format right-align
546 if (whole > 999) {
547 strcpy(distStr, "###.##"); // clamp
548 } else {
549 // Build the raw distance text
550 char tmp[8];
551 snprintf(tmp, sizeof(tmp), "%d.%02d", whole, frac);
552
553 // Measure its length
554 int len = (int)strlen(tmp);
555
556 // Compute how much padding is needed
557 // Ex. "1.23" = length 4 = pad 2
558 // "12.34" = length 5 = pad 1
559 int pad = 6 - len;
560 if (pad < 0) pad = 0;
561
562 // Fill in left padding
563 // Ex. pad = 2 -> " "
564 // pad = 1 -> " "
565 // pad = 0 -> ""
566 int i = 0;
567 while (i < pad) { distStr[i] = ' '; i++; }
568 distStr[i] = '\0';
569
570 // Append the actual number
571 // Ex. pad = 2 "1.23" -> " 1.23"
572 // pad = 1 "12.34" -> " 12.34"
573 // pad = 0 "123.45" -> "123.45"
574 strcat(distStr, tmp);
575 }
576 } else {
577 // If there is no valid reading, indicate no data
578 strcpy(distStr, " --.--");
579 }
580
581 // Clear row and lay out fixed columns
582 lcd.setCursor(0, 1); lcd.print(" ");
583 lcd.setCursor(0, 1); lcd.write((uint8_t)2); // [tank]
584 lcd.setCursor(1, 1); lcd.print(' ');
585 lcd.setCursor(2, 1); lcd.print(distStr); // DD.DD
586 lcd.setCursor(8, 1); lcd.print('c');
587 lcd.setCursor(9, 1); lcd.print('m');
588 lcd.setCursor(10,1); lcd.print(' ');
589 lcd.setCursor(11,1); lcd.write((uint8_t)3); // [pump]
590 lcd.setCursor(12,1); lcd.print(' ');
591 lcd.setCursor(13,1); // ON/OFF
592 if (s.pumpOn == 1) {
593 lcd.print("ON ");
594 } else {
595 lcd.print("OFF");
596 }
597
598 // Update "last drawn" cache
599 lastDist = s.distanceX100;
600 lastPumpOn = s.pumpOn;
601 }
602 }
603 }
604}

610void setup() {

613 pinMode(RELAY_PIN, OUTPUT);

616 pinMode(BTN_MANUAL_PIN, INPUT_PULLUP);
